# Выдать книгу
curl -X POST http://127.0.0.1:5000/api/give \
  -H "Content-Type: application/json" \
  -d '{"student_id": 1, "book_id": 1}'

# Сдать книгу
curl -X POST http://127.0.0.1:5000/api/return \
  -H "Content-Type: application/json" \
  -d '{"student_id": 1, "book_id": 1}'
